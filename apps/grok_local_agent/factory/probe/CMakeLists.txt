project(WARLOCKProbe LANGUAGES CXX)

  message(FATAL_ERROR "DPF is missing. Set DPF_DIR. Farnaz will not download it.")
endif()

  set(DPF_DIR $ENV{DPF_DIR})
endif()
add_subdirectory(${DPF_DIR} ${CMAKE_BINARY_DIR}/dpf)
